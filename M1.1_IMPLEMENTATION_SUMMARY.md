```python
class OpenAIGateway:
    """FastAPI server for OpenAI-compatible proxy."""
    
    def __init__(self, storage_backend, similarity_threshold=0.92, upstream_base_url=None, upstream_api_key=None):
        self.storage = storage_backend
        self.proxy = ProxyGateway(base_url=upstream_base_url, api_key=upstream_api_key)
        self.app = self._create_app()
    
    def _create_app(self) -> FastAPI:
        """Create FastAPI app with two routes:"""
        
        @app.get("/health")
        async def health_check() -> dict:
            return {"status": "ok", "service": "crucible-gateway"}
        
        @app.post("/v1/chat/completions")
        async def chat_completions(request: Request) -> Union[StreamingResponse, dict]:
            # Parse request JSON
            try:
                payload = await request.json()
            except json.JSONDecodeError as e:\n                raise HTTPException(\n                    status_code=400,\n                    detail=f"Invalid JSON: {str(e)}"
                )
            
            # Validate required fields
            if "messages" not in payload:
                raise HTTPException(status_code=400, detail="Missing 'messages' field")
            if "model" not in payload:
                raise HTTPException(status_code=400, detail="Missing 'model' field")
            
            is_streaming = payload.get("stream", False)
            
            # Try L1 exact match
            cache_key = payload_to_cache_key(payload)
            exact_match = await self.storage.get_by_hash(cache_key)
            
            if exact_match:
                logger.debug(f"Cache hit (exact): {cache_key[:8]}...")
                response = self._format_completion_response(
                    exact_match.response_text,
                    model=payload.get("model", ""),
                    finish_reason=exact_match.finish_reason,
                    usage={...}
                )
                if is_streaming:
                    return StreamingResponse(
                        self._format_sse_stream([response]),
                        media_type="text/event-stream"
                    )
                return response
            
            # Cache miss: relay to upstream
            logger.debug(f"Cache miss: {cache_key[:8]}...")
            
            if is_streaming:
                return StreamingResponse(
                    self._handle_streaming_request(payload),
                    media_type="text/event-stream"
                )
            return await self._handle_non_streaming_request(payload)
    
    async def _handle_non_streaming_request(self, payload) -> dict:
        """Relay non-streaming request to upstream"""
        response = await self.proxy.relay(payload, stream=False)
        if response.get("error"):
            raise HTTPException(status_code=response["error"]["status"], detail="Upstream error")
        return response
    
    async def _handle_streaming_request(self, payload) -> AsyncGenerator:
        """Relay streaming request with SSE passthrough"""
        async for chunk in self.proxy.relay_streaming(payload):
            yield chunk
    
    @staticmethod
    def _format_completion_response(content, model, finish_reason="stop", usage=None) -> dict:
        """Format response in OpenAI ChatCompletion format"""
        return {
            "id": "chatcmpl-cached",
            "object": "chat.completion",
            "created": 0,
            "model": model,
            "choices": [{
                "index": 0,
                "message": {"role": "assistant", "content": content},
                "finish_reason": finish_reason
            }],
            "usage": usage or {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0},
            "cache_hit": True
        }
    
    @staticmethod
    async def _format_sse_stream(responses) -> AsyncGenerator:
        """Format cached response as SSE stream"""
        for response in responses:
            yield f"data: {json.dumps(response)}\n\n"
        yield "data: [DONE]\n\n"
```

## Design: Proxy Gateway (infrastructure/proxy_gateway.py)

```python
class ProxyGateway:
    """Async HTTP client for upstream relay with streaming support."""
    
    def __init__(self, base_url=None, api_key=None, timeout=60.0, max_retries=2):
        self.base_url = base_url or "https://api.openai.com/v1"
        self.api_key = api_key
        self.timeout = timeout
        self.max_retries = max_retries
        self.client = httpx.AsyncClient(timeout=timeout)
    
    async def relay(self, payload, stream=False) -> dict:
        """Relay non-streaming request to upstream provider"""
        url = f"{self.base_url}/chat/completions"
        headers = self._build_headers(payload.get("model", ""))
        relay_payload = {**payload, "stream": False}
        
        for attempt in range(self.max_retries + 1):
            try:
                response = await self.client.post(url, json=relay_payload, headers=headers)
                if response.status_code == 200:
                    return response.json()
                
                # Handle errors
                error_data = self._parse_error_response(response)
                
                if should_fallback(response.status_code):
                    if attempt < self.max_retries:
                        wait_time = 2 ** attempt
                        logger.warning(f"Upstream error {response.status_code}, retrying in {wait_time}s")
                        await asyncio.sleep(wait_time)
                        continue
                
                # Non-retryable error
                logger.error(f"Upstream error {response.status_code}: {error_data}")
                return {
                    "error": {
                        "message": error_data.get("message", "Upstream error"),
                        "type": error_data.get("type", "api_error"),
                        "status": response.status_code
                    }
                }
            
            except httpx.TimeoutException:
                logger.error(f"Request timeout")
                if attempt < self.max_retries:
                    await asyncio.sleep(2 ** attempt)
                    continue
                return {
                    "error": {
                        "message": "Request timeout",
                        "type": "timeout_error",
                        "status": 504
                    }
                }
            except httpx.RequestError as e:\n                logger.error(f"Request error: {e}")
                return {
                    "error": {
                        "message": str(e),
                        "type": "request_error",
                        "status": 500
                    }
                }
        
        return {
            "error": {
                "message": "Max retries exceeded",
                "type": "api_error",
                "status": 503
            }
        }
    
    async def relay_streaming(self, payload) -> AsyncGenerator:
        """Relay streaming request to upstream provider with SSE passthrough"""
        url = f"{self.base_url}/chat/completions"
        headers = self._build_headers(payload.get("model", ""))
        relay_payload = {**payload, "stream": True}
        
        for attempt in range(self.max_retries + 1):
            try:
                logger.debug(f"Relaying streaming request to {url}")
                
                async with self.client.stream("POST", url, json=relay_payload, headers=headers) as response:
                    if response.status_code != 200:
                        error_data = await response.aread()
                        logger.error(f"Upstream streaming error {response.status_code}: {error_data}")
                        
                        if should_fallback(response.status_code) and attempt < self.max_retries:
                            await asyncio.sleep(2 ** attempt)
                            continue
                        
                        error_json = {
                            "error": {
                                "message": "Upstream streaming error",
                                "status": response.status_code
                            }
                        }
                        yield f"data: {json.dumps(error_json)}\n\n"
                        return
                    
                    # Stream SSE chunks
                    async for line in response.aiter_lines():
                        if line.strip():
                            yield f"{line}\n\n"
                
                return  # Success
            
            except httpx.TimeoutException:
                logger.error("Streaming request timeout")
                if attempt < self.max_retries:
                    await asyncio.sleep(2 ** attempt)
                    continue
                yield 'data: {"error": {"message": "Streaming timeout"}}\n\n'
                return
            except httpx.RequestError as e:\n                logger.error(f"Streaming request error: {e}")
                if attempt < self.max_retries:
                    await asyncio.sleep(2 ** attempt)
                    continue
                yield f'data: {{"error": {{"message": "{str(e)}"}}}}\n\n'
                return
    
    def _build_headers(self, model: str) -> dict:
        """Build HTTP headers for upstream request"""
        headers = {
            "Content-Type": "application/json",
            "User-Agent": "crucible-ai/0.1.0"
        }
        
        # Determine provider and set auth
        if "gpt" in model.lower():
            if self.api_key:
                headers["Authorization"] = f"Bearer {self.api_key}"
        elif "claude" in model.lower():
            if self.api_key:
                headers["x-api-key"] = self.api_key
        else:
            if self.api_key:
                headers["Authorization"] = f"Bearer {self.api_key}"
        
        return headers
    
    @staticmethod
    def _parse_error_response(response) -> dict:
        """Parse error response from upstream provider"""
        try:
            data = response.json()
            return data.get("error", {})
        except:
            return {"message": response.text or "Unknown error", "type": "api_error"}
    
    async def close(self):
        """Close HTTP client"""
        await self.client.aclose()
```
